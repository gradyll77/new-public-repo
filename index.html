import json import os import time import re import numpy as np import pandas as pd import yfinance as yf # ============================================================ # ALGO 7 — LEVERAGED ETF SCANNER # ============================================================ # # IMPORTANT: # The financial scoring engine below is intentionally preserved. # This version fixes the surrounding infrastructure: # # 1. ETF universe discovery # 2. 2X / 3X filtering # 3. inverse inclusion # 4. volume filtering # 5. bad/non-leveraged symbols # 6. duplicate symbols # 7. Yahoo/yfinance errors # 8. indentation/syntax issues # 9. invalid/missing market data # 10. JSON output for GitHub Pages/Blogger # # ============================================================ # ----------------------------- # SETTINGS # ----------------------------- MIN_DAILY_VOLUME = 300_000 HISTORY_PERIOD = "6mo" INCLUDE_INVERSE = True # Yahoo screener maximum is 250 results per request. SCREEN_PAGE_SIZE = 250 # Maximum number of ETF screener pages to inspect. # 20 x 250 = 5,000 ETF records. MAX_SCREEN_PAGES = 20 REQUEST_DELAY = 0.10 BREADTH_TICKER = "^GSPC" # ============================================================ # FALLBACK CORE LIST # ============================================================ # # These are only a safety net if Yahoo's ETF screener misses # something or is temporarily unavailable. # # IMPORTANT: # This list contains ETFs/funds intended to be 2X or 3X # leveraged products. Ordinary stocks such as COIN, HOOD, # MARA, IREN are intentionally NOT included. # # ============================================================ CORE_LEVERAGED_ETFS = [ # Semiconductors / technology "SOXL", "SOXS", "USD", # Nasdaq / technology "TQQQ", "SQQQ", "TECL", "TECS", "BULZ", "BERZ", "HIBL", "HIBS", "WEBL", "WEBS", # S&P / Dow / Russell "UPRO", "SPXU", "UDOW", "SDOW", "TNA", "TZA", "URTY", "SRTY", "MIDU", "MIDZ", "RETL", "REKT", # Energy "ERX", "ERY", "GUSH", "DRIP", "UCO", "SCO", "BOIL", "KOLD", "OILU", "OILD", # Gold / metals "AGQ", "ZSL", "NUGT", "DUST", "JNUG", "JDST", "GDXU", "GDXD", # Financials "FAS", "FAZ", # Healthcare "CURE", "PILL", # Real estate "DRN", "DRV", # Treasury "TMF", "TMV", # Volatility "UVIX", "UVXY", "SVIX", # Single-stock / thematic leveraged products "MSTU", "MSTX", "MSTP", "CONL", "BITX", "TSLL", "TSLQ", "NVDL", "NVD", "PTIR", "AMDL", "AMZU", "AMZZ", "FBL", "GGLL", "GGLS", "MSFL", "AAPU", "AAPD", "MUU", "MULL", "AVGU", # Other leveraged products "FNGU", "FNGD", "DUSL", "DOZR", "NAIL", ] # ============================================================ # SYMBOL EXCLUSIONS # ============================================================ # Ordinary stocks that were accidentally present in the old # manual list. They are NOT leveraged ETFs. EXCLUDED_SYMBOLS = { "COIN", "HOOD", "MARA", "IREN", } # ============================================================ # LEVERAGE DETECTION # ============================================================ def is_2x_or_3x_leveraged_name(symbol, name): """ Determine whether Yahoo's ETF name appears to represent a 2X or 3X leveraged product. We intentionally do NOT simply search for 'BULL' or 'BEAR', because that can catch products that are not actually 2X/3X. """ text = f"{symbol} {name}".upper() # Explicit 2X / 3X descriptions patterns = [ r"\b2X\b", r"\b3X\b", r"\b2X LONG\b", r"\b3X LONG\b", r"\b2X SHORT\b", r"\b3X SHORT\b", r"\b2X BULL\b", r"\b3X BULL\b", r"\b2X BEAR\b", r"\b3X BEAR\b", ] for pattern in patterns: if re.search(pattern, text): return True return False def is_inverse_name(symbol, name): """ Identify products that are explicitly inverse/short/bear. """ text = f"{symbol} {name}".upper() inverse_terms = [ "BEAR", "SHORT", "INVERSE", "-2X", "-3X", ] return any(term in text for term in inverse_terms) # ============================================================ # YAHOO ETF SCREENER # ============================================================ def get_yahoo_etf_universe(): """ Pull U.S. ETFs from Yahoo's ETF screener in pages. We do NOT use dayvolume as the ETF screener sort field. Volume is verified later from historical price data. Yahoo limits a screener response to 250 records, so we paginate through multiple pages. """ print("\nDownloading Yahoo ETF universe...") discovered = {} try: query = yf.ETFQuery( "and", [ yf.ETFQuery( "eq", ["region", "us"] ) ] ) for page in range(MAX_SCREEN_PAGES): offset = page * SCREEN_PAGE_SIZE print( f" Yahoo ETF page {page + 1} " f"(offset {offset})..." ) response = yf.screen( query, offset=offset, size=SCREEN_PAGE_SIZE, sortField="ticker", sortAsc=True ) quotes = response.get("quotes", []) if not quotes: print(" No more Yahoo ETF records.") break for item in quotes: symbol = str( item.get("symbol", "") ).upper().strip() name = ( item.get("longName") or item.get("shortName") or "" ) if not symbol: continue if symbol in EXCLUDED_SYMBOLS: continue if not is_2x_or_3x_leveraged_name( symbol, name ): continue if ( not INCLUDE_INVERSE and is_inverse_name(symbol, name) ): continue discovered[symbol] = { "symbol": symbol, "name": name, "source": "Yahoo ETF Screener" } # If Yahoo returned fewer than the requested page # size, we've reached the end. if len(quotes) < SCREEN_PAGE_SIZE: break time.sleep(REQUEST_DELAY) except Exception as e: print( "Yahoo ETF screener unavailable:", repr(e) ) print( f"Yahoo discovered " f"{len(discovered)} possible 2X/3X ETFs." ) return discovered # ============================================================ # BUILD COMPLETE UNIVERSE # ============================================================ def build_universe(): yahoo_candidates = get_yahoo_etf_universe() symbols = set() # Add Yahoo candidates for symbol in yahoo_candidates: if symbol not in EXCLUDED_SYMBOLS: symbols.add(symbol) # Add safety-net core list for symbol in CORE_LEVERAGED_ETFS: if symbol not in EXCLUDED_SYMBOLS: symbols.add(symbol) symbols = sorted(symbols) print( f"\nTotal candidate symbols before " f"volume validation: {len(symbols)}" ) return symbols # ============================================================ # DOWNLOAD DAILY DATA SAFELY # ============================================================ def get_history(ticker, period=HISTORY_PERIOD): try: df = yf.Ticker(ticker).history( period=period, interval="1d", auto_adjust=False ) if df is None or df.empty: return None # Required columns required_columns = [ "Open", "High", "Low", "Close", "Volume", ] for column in required_columns: if column not in df.columns: return None # Remove rows without essential data df = df.dropna( subset=[ "High", "Low", "Close", "Volume" ] ) if df.empty: return None return df except Exception as e: print( f" {ticker}: history error -> {e}" ) return None # ============================================================ # VOLUME FILTER # ============================================================ def volume_filter(tickers): qualified = [] print("\nChecking daily volume...") for ticker in tickers: df = get_history( ticker, period="1mo" ) if df is None or df.empty: continue if len(df) < 5: continue try: latest_volume = int( df["Volume"].iloc[-1] ) avg_volume_5d = float( df["Volume"] .tail(5) .mean() ) if latest_volume >= MIN_DAILY_VOLUME: qualified.append( { "symbol": ticker, "daily_volume": latest_volume, "avg_volume_5d": round( avg_volume_5d ) } ) print( f" QUALIFIED " f"{ticker:<8} " f"volume={latest_volume:,}" ) except Exception as e: print( f" {ticker}: volume error -> {e}" ) time.sleep(REQUEST_DELAY) return qualified # ============================================================ # MARKET BREADTH # ============================================================ def get_market_breadth(): try: spy = yf.Ticker( BREADTH_TICKER ).history( period="1y", interval="1d" ) if spy is None or spy.empty: return "NEUTRAL", 1.0 if len(spy) < 200: return "NEUTRAL", 1.0 current_price = float( spy["Close"].iloc[-1] ) sma200 = float( spy["Close"] .rolling(200) .mean() .iloc[-1] ) if current_price > sma200: return "BULLISH", 1.15 return "BEARISH", 0.85 except Exception as e: print( "Market breadth error:", e ) return "NEUTRAL", 1.0 # ============================================================ # ALGO 7 FINANCIAL ENGINE # ============================================================ # # DO NOT CHANGE THIS SECTION IF YOU WANT TO KEEP THE ORIGINAL # ALGO 7 ENGINE. # # ============================================================ def calculate_algo_score( df, breadth_multiplier ): if len(df) < 50: return ( 0, "DISQUALIFIED", "Insufficient Data" ) df = df.copy() df["Typical_Price"] = ( df["High"] + df["Low"] + df["Close"] ) / 3 df["PV"] = ( df["Typical_Price"] * df["Volume"] ) vwap_20 = ( df["PV"].rolling(20).sum() / df["Volume"].rolling(20).sum() ) current_vwap = float( vwap_20.iloc[-1] ) current_close = float( df["Close"].iloc[-1] ) sma20 = float( df["Close"] .rolling(20) .mean() .iloc[-1] ) sma50 = float( df["Close"] .rolling(50) .mean() .iloc[-1] ) base_score = 0 # -------------------------------- # VWAP # -------------------------------- if current_close > current_vwap: vwap_dist = ( current_close - current_vwap ) / current_vwap if 0.005 <= vwap_dist <= 0.08: base_score += 35 elif vwap_dist > 0.08: base_score += 20 else: base_score += 25 else: base_score += 5 # -------------------------------- # TREND # -------------------------------- if current_close > sma20 > sma50: base_score += 35 elif current_close > sma20: base_score += 20 elif current_close > sma50: base_score += 10 # -------------------------------- # VOLUME # -------------------------------- vol_sma20 = float( df["Volume"] .rolling(20) .mean() .iloc[-1] ) latest_volume = float( df["Volume"].iloc[-1] ) if latest_volume > vol_sma20 * 1.2: base_score += 30 elif latest_volume > vol_sma20: base_score += 15 # -------------------------------- # MARKET REGIME # -------------------------------- final_score = int( round( base_score * breadth_multiplier ) ) final_score = max( 0, min(100, final_score) ) # -------------------------------- # GRADE # -------------------------------- if final_score >= 75: grade = "HIGH CONVICTION" status = "BULLISH LEAD" elif final_score >= 50: grade = "MODERATE" status = "WATCHLIST" else: grade = "WEAK" status = "AVOID / CASH" return ( final_score, grade, status ) # ============================================================ # SCAN INDIVIDUAL ETF # ============================================================ def analyze_ticker( ticker, breadth_multiplier ): df = get_history( ticker, period=HISTORY_PERIOD ) if df is None: return None if len(df) < 50: return None try: score, grade, status = ( calculate_algo_score( df, breadth_multiplier ) ) latest_close = float( df["Close"].iloc[-1] ) avg_volume_20 = float( df["Volume"] .tail(20) .mean() ) latest_volume = int( df["Volume"].iloc[-1] ) # 5-day momentum if len(df) >= 6: momentum_5d = ( ( df["Close"].iloc[-1] / df["Close"].iloc[-6] ) - 1 ) * 100 else: momentum_5d = np.nan sma20 = float( df["Close"] .rolling(20) .mean() .iloc[-1] ) distance_sma20 = ( ( latest_close / sma20 ) - 1 ) * 100 return { "symbol": ticker, "score": int(score), "grade": grade, "status": status, "price": round( latest_close, 2 ), "volume": latest_volume, "avg_volume_20d": round( avg_volume_20 ), "momentum_5d_pct": round( float(momentum_5d), 2 ), "distance_sma20_pct": round( float(distance_sma20), 2 ) } except Exception as e: print( f" {ticker}: analysis error -> {e}" ) return None # ============================================================ # MAIN PIPELINE # ============================================================ def run_pipeline(): print("=" * 70) print("ALGO 7 — LEVERAGED ETF MARKET SCANNER") print("=" * 70) # -------------------------------- # 1. Build universe # -------------------------------- universe = build_universe() # -------------------------------- # 2. Volume filter # -------------------------------- qualified = volume_filter( universe ) qualified_symbols = [ item["symbol"] for item in qualified ] print( "\n" f"Qualified funds over " f"{MIN_DAILY_VOLUME:,} " f"latest-day volume: " f"{len(qualified_symbols)}" ) # -------------------------------- # 3. Market regime # -------------------------------- ( breadth_regime, breadth_mult ) = get_market_breadth() print( "\nMarket regime:", breadth_regime ) print( "Breadth multiplier:", breadth_mult ) # -------------------------------- # 4. Analyze qualified ETFs # -------------------------------- results = [] print( "\nRunning Algo 7..." ) for ticker in qualified_symbols: print( f" Analyzing {ticker}..." ) result = analyze_ticker( ticker, breadth_mult ) if result is not None: results.append(result) time.sleep( REQUEST_DELAY ) # -------------------------------- # 5. Sort leaderboard # -------------------------------- results.sort( key=lambda x: ( x["score"], x["momentum_5d_pct"] ), reverse=True ) # -------------------------------- # 6. Assign ranks # -------------------------------- for i, item in enumerate( results, start=1 ): item["rank"] = i # -------------------------------- # 7. Output # -------------------------------- output_payload = { "scanner": "Leveraged ETF Algo 7", "engine_version": "Algo 7 Original Engine", "minimum_volume": MIN_DAILY_VOLUME, "history_period": HISTORY_PERIOD, "include_inverse": INCLUDE_INVERSE, "universe_size": len(universe), "volume_qualified": len(qualified_symbols), "qualified_funds": len(results), "market_regime": breadth_regime, "breadth_multiplier": breadth_mult, "last_updated": pd.Timestamp.now( tz="UTC" ).strftime( "%Y-%m-%d %H:%M UTC" ), "tickers": results } # -------------------------------- # 8. Save JSON # -------------------------------- os.makedirs( "docs", exist_ok=True ) output_files = [ "docs/screener_data.json", "screener_data.json" ] for filename in output_files: with open( filename, "w", encoding="utf-8" ) as f: json.dump( output_payload, f, indent=2 ) # -------------------------------- # 9. Console leaderboard # -------------------------------- print("\n") print("=" * 70) print("ALGO 7 LEADERBOARD") print("=" * 70) for item in results: print( f"{item['rank']:>3}. " f"{item['symbol']:<8} " f"{item['score']:>3} " f"{item['grade']:<18} " f"{item['status']}" ) print("=" * 70) print( f"\nSaved {len(results)} results." ) print( "JSON files created:" ) for filename in output_files: print( f" {filename}" ) return output_payload # ============================================================ # START # ============================================================ if __name__ == "__main__": run_pipeline()
