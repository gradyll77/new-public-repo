(100 / (1 + rs)) return float(rsi.iloc[-1]) def fetch_market_data(ticker: str): """Fetches daily prices via yfinance and computes SMA10, SMA20, and RSI14.""" try: data = yf.Ticker(ticker).history(period="2mo") if data.empty or len(data) < 20: return None close = data["Close"] price = float(close.iloc[-1]) sma10 = float(close.rolling(window=10).mean().iloc[-1]) sma20 = float(close.rolling(window=20).mean().iloc[-1]) rsi = round(calculate_rsi(close, 14), 1) return {"price": price, "sma10": sma10, "sma20": sma20, "rsi": rsi} except Exception as e: print(f"Error fetching data for {ticker}: {e}") return None # ============================================================================== # REVERSE-ENGINEERED SIDEKICK & CONFLICT EVALUATION # ============================================================================== def evaluate_symbol(ticker: str): """Evaluates a ticker using reverse-engineered Sidekick rules (75% weight) and StockInvest score (25% weight) with conflict disqualification. """ tech = fetch_market_data(ticker) if not tech: return None price = tech["price"] sma10 = tech["sma10"] sma20 = tech["sma20"] rsi = tech["rsi"] si_score = STOCKINVEST_SCORES.get(ticker, 0.0) # 1. Reverse-Engineered Sidekick Criteria (Max Scale: -10 to +10) sk_score = 0.0 # Bullish Trend Checks if price > sma10 and sma10 > sma20: sk_score += 5.0 elif price > sma10: sk_score += 2.0 # RSI Momentum Sweet Spot (55.0 to 67.0) if 55.0 <= rsi <= 67.0: sk_score += 3.5 elif 50.0 <= rsi < 55.0 or 67.0 < rsi <= 70.0: sk_score += 1.5 # Price Clearance above SMA10 if price > sma10: clearance_pct = ((price - sma10) / sma10) * 100 if clearance_pct >= 1.0: sk_score += 1.5 # Bearish Trend Checks if rsi <= 25.5: sk_score -= 6.0 elif rsi <= 31.0: sk_score -= 4.0 if price < sma10 and sma10 < sma20: sk_score -= 4.0 # Clamp Sidekick score between -10.0 and +10.0 sk_score = max(min(sk_score, 10.0), -10.0) # 2. Weighted Score Calculation final_score = round((sk_score * 0.75) + (si_score * 0.25), 2) # 3. Conflict and Disqualification Check is_sk_buy = sk_score > 2.0 is_sk_sell = sk_score < -2.0 is_si_buy = si_score > 0.0 is_si_sell = si_score < 0.0 has_conflict = (is_sk_buy and is_si_sell) or (is_sk_sell and is_si_buy) return { "ticker": ticker, "score": final_score, "sk_score": sk_score, "si_score": si_score, "rsi": rsi, "price": price, "sma10": sma10, "has_conflict": has_conflict, } def generate_screener_data(): """Runs through the ETF universe and categorizes into Buys, Sells, and Holds.""" buys, sells, holds = [], [], [] for ticker in ETF_UNIVERSE: res = evaluate_symbol(ticker) if not res: continue setup_text = f"Price ${res['price']:.2f} | Sidekick: {res['sk_score']} | SI: {res['si_score']}" item = { "ticker": res["ticker"], "score": res["score"], "isBear": res["score"] < 0, "setup": setup_text, "rsi": res["rsi"], } # Conflict override: push divergent picks to Holds if res["has_conflict"]: item["setup"] = ( f"⚠️ CONFLICT OVERRIDE (SI Score: {res['si_score']})" ) holds.append(item) elif res["score"] > 1.5 and res["si_score"] >= 0: buys.append(item) elif res["score"] < -1.5 and res["si_score"] <= 0: sells.append(item) else: holds.append(item) # Sort results buys.sort(key=lambda x: x["score"], reverse=True) sells.sort(key=lambda x: x["score"]) # Most negative magnitude first return buys[:5], sells[:5], holds # ============================================================================== # HTML BUILDER & FILE OUTPUT # ============================================================================== def update_html_file(buys, sells, holds, filepath="index.html"): """Reads index.html and updates the JS data arrays with fresh weighted scores.""" if not os.path.exists(filepath): print(f"File {filepath} not found.
